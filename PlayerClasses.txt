enum invSortState{
	all,
	weapons,
	ammo,
};
class CustomPlayerPawn : DoomPlayer
{
	double totalWeight, maxWeight;
	invSortState sortState; //which type of items should be displayed
	int selectionIndex; //last selected inventroy item
	Default {
		Player.DisplayName "Inventory test"; 
		//Player.AttackZOffset 13; 
		//Player.ViewHeight 64; 
	}
	
    override void PostBeginPlay(){
        Super.PostBeginPlay();
		sortState = all;
		selectionIndex = 0;
		maxWeight = 10.0;
		Inventory item = FindInventory("HexenArmor"); if (item != null){RemoveInventory(item);}
		totalWeight = CalcWeight();
	}
	
	double CalcWeight() {
		double weight = 0;
		for (Inventory item = Inv; item != null; item = item.Inv) {
			HasWeight weightBehavior = HasWeight(item.FindBehavior("HasWeight"));
			if (weightBehavior) { weight += (item.Amount * weightBehavior.weight); }
		}
		return weight;
	}
	
	 override bool CanTouchItem(Inventory item){
		totalWeight = CalcWeight();
		HasWeight weightBehavior = HasWeight(item.FindBehavior("HasWeight"));
		if (weightBehavior != null){
			if (totalWeight + weightBehavior.weight > maxWeight){ return false; }
			else{ totalWeight += weightBehavior.weight; }
		}
		return Super.CanTouchItem(item);
	 }
	 
	 
	 override int DamageMobj(Actor inflictor, Actor source, int dam, Name mod, int flags , double angle ){
		dam += 30;
		Super.DamageMobj(inflictor, source, dam, mod, flags, angle);
		return dam;
	 }
	 
	 override void SpawnLineAttackBlood(
        Actor attacker,
        Vector3 bleedpos,
        double SrcAngleFromTarget,
        int originaldamage,
        int actualdamage
    )
    {
		super.SpawnLineAttackBlood(attacker, bleedpos, SrcAngleFromTarget, originaldamage, actualdamage);
	
	
	}
	 
	 
}

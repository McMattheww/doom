enum invSortState{
	all,
	weapons,
	ammo,
};
class CustomPlayerPawn : PlayerPawn
{
	double totalWeight, maxWeight;
	invSortState sortState; //which type of items should be displayed
	int selectionIndex; //last selected inventroy item
	Default { Player.DisplayName "Inventory test"; }
	
    override void PostBeginPlay(){
        Super.PostBeginPlay();
		sortState = all;
		selectionIndex = 0;
		maxWeight = 5.0;
		Inventory item = FindInventory("HexenArmor"); if (item != null){RemoveInventory(item);}
		totalWeight = CalcWeight();
	}
	
	double CalcWeight() {
		double weight = 0;
		for (Inventory item = Inv; item != null; item = item.Inv) {
			HasWeight weightBehavior = HasWeight(item.FindBehavior("HasWeight"));
			if (weightBehavior) { weight += (item.Amount * weightBehavior.weight); }
		}
		return weight;
	}
	
	 override bool CanTouchItem(Inventory item){
		totalWeight = CalcWeight();
		HasWeight weightBehavior = HasWeight(item.FindBehavior("HasWeight"));
		if (weightBehavior != null){
			if (totalWeight + weightBehavior.weight > maxWeight){ return false; }
			else{ totalWeight += weightBehavior.weight; }
		}
		return Super.CanTouchItem(item);
	 }

		return dam;

}

Class InventoryHandler : EventHandler {
	override void NetworkProcess(ConsoleEvent e){
		CustomPlayerPawn p = CustomPlayerPawn(players[e.player].mo);
		if (e.name == "inventoryOpen"){ Menu.SetMenu("InventoryMenu"); }
		else if (e.name == "saveIndex"){ p.selectionIndex = e.Args[0]; }
		//if no specific event name identified, event name is used as String argument to drop specific item
		else { p.A_DropInventory(e.name, 1); }
	}
}

class ZombieClone : ZombieMan {
	Headshottable hbox;
	override void PostBeginPlay(){
		super.PostBeginPlay();
		hbox = Headshottable(AddBehavior("Headshottable"));
		hbox.origin = (0, 39);
		hbox.height = 20;
		hbox.radius = 7;
		hbox.multiplier = 2.0;
	}
	Default
    {
        Health 1000;
    }
	override int DamageMobj(Actor inflictor,Actor source,int dam,Name mod,int flags,double angle){
		Headshottable head = Headshottable(FindBehavior("Headshottable"));
				if (head != null && mod != "radius"){
					double lowerY = pos.z + head.origin.y;
					double upperY = lowerY + head.height;
					if (inflictor.pos.z > lowerY && inflictor.pos.z < upperY){
						Console.Printf("Headshot!");
						dam *= head.multiplier;
					}
				}
		
		Super.DamageMobj(inflictor, source, dam, mod, flags, angle);
		console.printf("damage");
		return dam;
	
	}
}

Class InventoryHandler : EventHandler {
	override void NetworkProcess(ConsoleEvent e){
		CustomPlayerPawn p = CustomPlayerPawn(players[e.player].mo);
		if (e.name == "inventoryOpen"){
			//p.A_DropInventory ("testobj", 1);
			//p.frozen = !p.frozen;
			Menu.SetMenu("InventoryMenu");
		}
		if (e.name == "inventoryUp"){
			p.selectionIndex ++;

		}
		if (e.name == "inventoryDown"){
			p.selectionIndex --;
		}
		if (e.name == "saveIndex"){
			p.selectionIndex = e.Args[0];
		}
		else { //if no specific event name identified, event name is used as String argument to drop specific item
			p.A_DropInventory(e.name, 1);
		}
	}
}

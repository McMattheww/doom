
class BreakableWeapon : DoomWeapon{
	Array<double> durabilityList;
	double durability;
	int active;
	HasWeight weightBehavior;
	
	Default{
		inventory.MaxAmount 20;
	}
	
	
	override void PostBeginPlay(){
		super.PostBeginPlay();
		durability = 100.0;
		active = 0;
		weightBehavior = HasWeight(AddBehavior("HasWeight"));
	}
	
	override void AttachToOwner(Actor other){
		durabilityList.Push(durability);
		Super.AttachToOwner(other);
	}
	
	override bool HandlePickup (Inventory item) {
		bool res = Super.HandlePickup(item);
		if (res){
			durabilityList.Push(durability);
			for (int i = 0; i < durabilityList.Size(); i++){
			Console.Printf("durability[%d] = %f, size = %d",i,durabilityList[i],durabilityList.Size());
			}
		}
		return res;
	}

	override void Tick(){
		super.tick();
		if (Owner && amount == 0){
			if (Owner.player.ReadyWeapon.GetClass() != self.GetClass()){Destroy();}
		}
	}

	
	
		
	action state A_CheckBreak(StateLabel label){
		BreakableWeapon ready = BreakableWeapon(player.ReadyWeapon);
		ready.durabilityList[ready.active] -= 20.0;
		ready.durability = ready.durabilityList[ready.active];
		Console.Printf("%f", ready.durabilityList[ready.active]);
		
		if (ready.durabilityList[ready.active] <= 0.0){
			ready.durabilityList.Delete(ready.active);
			ready.active = 0;
			ready.Amount--;
			return ResolveState(label);
		}
		return null;
	}
	
	
	action void A_NewWeapon(){
		if (player.ReadyWeapon.amount > 0){
			player.PendingWeapon = player.mo.PickNewWeapon(null);
		}
		else{
			player.PendingWeapon = weapon(player.mo.FindInventory("Fist"));
		}	
	}
	
	
	override Inventory CreateTossable (int amt){
		// If self actor lacks a SpawnState, don't drop it. (e.g. A base weapon
		// like the fist can't be dropped because you'll never see it.)
		if (SpawnState == GetDefaultByType("Actor").SpawnState || SpawnState == NULL){
			return NULL;
		}
		if (bUndroppable || bUntossable || Owner == NULL || Amount <= 0 || amt == 0){
			return NULL;
		}
		amt = clamp(amt, 1, Amount);
		if (Amount == amt && !bKeepDepleted){
			BecomePickup ();
			DropTime = 30;
			bSpecial = bSolid = false;
			return self;
		}
		let copy = Inventory(Spawn (GetClass(), Owner.Pos, NO_REPLACE));
		if (copy != NULL){
			BreakableWeapon wep = BreakableWeapon(copy);
			if (wep!= NULL){
				wep.durabilityList.Delete(wep.active);
				wep.active = 0;
			}
			copy.MaxAmount = MaxAmount;
			copy.Amount = amt;
			copy.DropTime = 30;
			copy.bSpecial = copy.bSolid = false;
			Amount -= amt;
		}
		return copy;
	}

	override void PostBeginPlay(){
		super.PostBeginPlay();
		if (durability.size() == 0){durability.push(100.0);}
		active = 0;
		weightBehavior = HasWeight(AddBehavior("HasWeight"));
	}
	
	
	override bool HandlePickup (Inventory item) {
		bool res = Super.HandlePickup(item);
		if (res){
			BreakableWeapon wep = BreakableWeapon(item);
			durability.Push(wep.durability[0]);
		}
		return res;
	}

	override void Tick(){
		super.tick();
		if (Owner && amount == 0){
			if (Owner.player.ReadyWeapon.GetClass() != self.GetClass()){Destroy();}
		}
	}

	//call after weapon fires to reduce durability and go to break state if durability is less than 0
	action state A_CheckBreak(StateLabel label){
		BreakableWeapon ready = BreakableWeapon(player.ReadyWeapon);
		ready.durability[ready.active] -= 20.0;
		Console.Printf("%f", ready.durability[ready.active]);
		for (int i = 0; i < ready.durability.Size(); i++){
			Console.Printf("durability[%d] = %f, size = %d",i,ready.durability[i],ready.durability.Size());
			}
		if (ready.durability[ready.active] <= 0.0){
			ready.durability.Delete(ready.active);
			ready.active = 0;
			ready.Amount--;
			return ResolveState(label);
		}
		return null;
	}
	
	//call during break state to switch selected weapon
	action void A_NewWeapon(){
		if (player.ReadyWeapon.amount > 0){
			player.PendingWeapon = player.mo.PickNewWeapon(null);
		}
		else{
			player.PendingWeapon = weapon(player.mo.FindInventory("Fist"));
		}	
	}
	
	
	action void A_LoadClip(){
		Class<Inventory> magazine = player.ReadyWeapon.AmmoType1;
		Class<Inventory> source = player.ReadyWeapon.AmmoType2;
		Inventory m = player.mo.FindInventory(magazine.GetClassName());
		Inventory s = player.mo.FindInventory(source.GetClassName());
		int diff = m.maxamount - m.amount;
		player.mo.A_GiveInventory(magazine, diff);
		player.mo.A_TakeInventory(source, diff);
	}
	
	action void A_CycleAmmo(){
		Weapon wep = player.ReadyWeapon;
		BreakableWeapon bwep = BreakableWeapon(wep);
		Class<Inventory> magazine = player.ReadyWeapon.AmmoType1;
		Class<Inventory> source = player.ReadyWeapon.AmmoType2;
		Inventory m = player.mo.FindInventory(magazine.GetClassName());
		Inventory s = player.mo.FindInventory(source.GetClassName());
		int remaining = m.amount; m.amount = 0;
		s.amount += remaining;
		bwep.activeAmmo += 1;
		bwep.activeAmmo %= bwep.ammoTypes;
	}

	override bool CheckAmmo(int fireMode, bool autoSwitch, bool requireAmmo, int ammocount)
	{
		int count1, count2;
		int enough, enoughmask;
		int lAmmoUse1;
		int lAmmoUse2 = AmmoUse2;

		if (sv_infiniteammo || (Owner.FindInventory ('PowerInfiniteAmmo', true) != null))
		{
			return true;
		}
		if (fireMode == EitherFire)
		{
			bool gotSome = CheckAmmo (PrimaryFire, false) || CheckAmmo (AltFire, false);
			if (!gotSome && autoSwitch)
			{
				PlayerPawn(Owner).PickNewWeapon (null);
			}
			return gotSome;
		}
		let altFire = (fireMode == AltFire);
		let optional = (altFire? bAlt_Ammo_Optional : bAmmo_Optional);
		let useboth = (altFire? bAlt_Uses_Both : bPrimary_Uses_Both);

		if (!requireAmmo && optional)
		{
			return true;
		}
		count1 = (Ammo1 != null) ? Ammo1.Amount : 0;
		count2 = (Ammo2 != null) ? Ammo2.Amount : 0;

		if (bDehAmmo && Ammo1 == null)
		{
			lAmmoUse1 = 0;
		}
		else if (ammocount >= 0)
		{
			lAmmoUse1 = ammocount;
			lAmmoUse2 = ammocount;
		}
		else
		{
			lAmmoUse1 = AmmoUse1;
		}

		enough = (count1 >= lAmmoUse1) | ((count2 >= lAmmoUse2) << 1);
		if (useboth)
		{
			enoughmask = 3;
		}
		else
		{
			enoughmask = 1 << altFire;
		}
		if (altFire && FindState('AltFire') == null)
		{ // If this weapon has no alternate fire, then there is never enough ammo for it
			enough &= 1;
		}
		if (((enough & enoughmask) == enoughmask) || (enough && bAmmo_CheckBoth))
		{
			return true;
		}
		// out of ammo, pick a weapon to change to
		if (autoSwitch)
		{
			PlayerPawn(Owner).PickNewWeapon (null);
		}
		return false;
	}
	
	
	override Inventory CreateTossable (int amt){
		// If self actor lacks a SpawnState, don't drop it. (e.g. A base weapon
		// like the fist can't be dropped because you'll never see it.)
		if (SpawnState == GetDefaultByType("Actor").SpawnState || SpawnState == NULL){
			return NULL;
		}
		if (bUndroppable || bUntossable || Owner == NULL || Amount <= 0 || amt == 0){
			return NULL;
		}
		amt = clamp(amt, 1, Amount);
		if (Amount == amt && !bKeepDepleted){
			BecomePickup ();
			DropTime = 30;
			bSpecial = bSolid = false;
			return self;
		}
		//when dropping breakable weapon with another remainnig in inventory, remove from durability list
		let copy = BreakableWeapon(Spawn (GetClass(), Owner.Pos, NO_REPLACE));
		if (copy != NULL){
			inventory wep = Owner.FindInventory(GetClassName());
			BreakableWeapon bwep = BreakableWeapon(wep);
			if (bwep!= NULL){
				copy.durability.push(bwep.durability[bwep.active]);
				bwep.durability.Delete(bwep.active);
				bwep.active = 0;
			}
			copy.MaxAmount = MaxAmount;
			copy.Amount = amt;
			copy.DropTime = 30;
			copy.bSpecial = copy.bSolid = false;
			Amount -= amt;
		}
		return inventory(copy);
	}
	

}
